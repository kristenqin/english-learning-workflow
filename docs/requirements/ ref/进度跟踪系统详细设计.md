### 1.1 系统目标

def smooth_progress_data(raw_data, window_size=7):
    """
    使用移动平均算法平滑进度数据
    """
    smoothed_data = []
    
    for i in range(len(raw_data)):
        start_idx = max(0, i - window_size // 2)
        end_idx = min(len(raw_data), i + window_size // 2 + 1)
        
        window_average = sum(raw_data[start_idx:end_idx]) / (end_idx - start_idx)
        smoothed_data.append(window_average)
    
    return smoothed_data
```

def calculate_ranking_score(user_id, ranking_type, time_period):
    """
    计算用户在特定排行榜中的分数
    """
    base_metrics = get_user_metrics(user_id, time_period)
    
    if ranking_type == 'overall':
        score = (
            base_metrics['average_score'] * 0.4 +
            base_metrics['practice_frequency'] * 0.3 +
            base_metrics['improvement_rate'] * 0.3
        )
    elif ranking_type == 'activity':
        score = (
            base_metrics['total_practice_time'] * 0.5 +
            base_metrics['days_active'] * 0.3 +
            base_metrics['level_completions'] * 0.2
        )
    elif ranking_type == 'progress':
        score = (
            base_metrics['skill_improvement'] * 0.6 +
            base_metrics['goal_achievement_rate'] * 0.4
        )
    
    return round(score, 2)
```
